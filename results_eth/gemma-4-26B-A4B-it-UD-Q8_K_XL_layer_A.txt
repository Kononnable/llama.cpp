class EventBus {
private:
    struct Handler {
        int id;
        std::function<void(const std::string&)> callback;
        int priority;
    };

    std::unordered_map<std::string, std::vector<Handler>> subscribers;
    std::mutex mtx;
    int nextId = 0;
    bool running = true;
    std::vector<std::thread> workers;
    std::vector<std::pair<std::string, std::string>> pendingEvents;

    void processEvents() {
        while (running) {
            std::vector<std::pair<std::string, std::string>> localEvents;
            {
                std::lock_guard<std::mutex> lock(mtx);
                localEvents.swap(pendingEvents);
            }
            
            if (localEvents.empty()) {
                std::this_thread::sleep_for(std::chrono::milliseconds(1));
                continue;
            }
            
            for (const auto& [eventType, payload] : localEvents) {
                auto it = subscribers.find(eventType);
                if (it != subscribers.end()) {
                    auto handlers = it->second;
                    std::sort(handlers.begin(), handlers.end(), 
                        [](const Handler& a, const Handler& b) {
                            return a.priority > b.priority;
                        });
                    for (const auto& handler : handlers) {
                        handler.callback(payload);
                    }
                }
            }
        }
    }

public:
    EventBus(int numWorkers = 4) {
        for (int i = 0; i < numWorkers; ++i) {
            workers.emplace_back(&EventBus::processEvents, this);
        }
    }
    
    ~EventBus() {
        running = false;
        for (auto& worker : workers) {
            if (worker.joinable()) worker.join();
        }
    }
    
    int subscribe(const std::string& eventType, 
                  std::function<void(const std::string&)> callback,
                  int priority = 0) {
        std::lock_guard<std::mutex> lock(mtx);
        int id = nextId++;
        subscribers[eventType].push_back({id, callback, priority});
        return id;
    }
    
    void unsubscribe(const std::string& eventType, int id) {
        std::lock_guard<std::mutex> lock(mtx);
        auto it = subscribers.find(eventType);
        if (it != subscribers.end()) {
            auto& handlers = it->second;
            handlers.erase(std::remove_if(handlers.begin(), handlers.end(),
                [id](const Handler& h) { return h.id == id; }),
                handlers.end());
        }
    }
    
    void emit(const std::string& eventType, const std::string& payload) {
        std::lock_guard<std::mutex> lock(mtx);
        pendingEvents.emplace_back(eventType, payload);
    }
    
    size_t getPendingCount() {
        std::lock_guard<std::mutex> lock(mtx);
        return pendingEvents.size();
    }
};

class DataProcessor {
private:
    std::vector<int> data;
    mutable std::mutex dataMutex;
    EventBus& bus;
    int eventCounter = 0;
    
public:
    DataProcessor(EventBus& eventBus) : bus(eventBus) {
        bus.subscribe("data.input", [this](const std::string& payload) {
            std::lock_guard<std::mutex> lock(dataMutex);
            int value = std::stoi(payload);
            data.push_back(value);
            eventCounter++;
            bus.emit("data.processed", std::to_string(value * 2));
        }, 10);
        
        bus.subscribe("data.clear", [this](const std::string&) {
            std::lock_guard<std::mutex> lock(dataMutex);
            data.clear();
            bus.emit("data.cleared", "All data removed");
        });
    }
    
    std::vector<int> getData() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        return data;
    }
    
    double calculateAverage() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        if (data.empty()) return 0.0;
        long long sum = 0;
        for (int val : data) {
            sum += val;
        }
        return static_cast<double>(sum) / data.size();
    }
    
    int getEventCount() const { return eventCounter; }
};

class MetricsCollector {
private:
    struct Metric {
        std::string name;
        double value;
        std::chrono::system_clock::time_point timestamp;
    };
    std::vector<Metric> metrics;
    mutable std::mutex mtx;
    
public:
    void record(const std::string& name, double value) {
        std::lock_guard<std::mutex> lock(mtx);
        metrics.push_back({name, value, std::chrono::system_clock::now()});
    }
    
    void printSummary() const {
        std::lock_guard<std::mutex> lock(mtx);
        std::unordered_map<std::string, double> averages;
        std::unordered_map<std::string, int> counts;
        
        for (const auto& m : metrics) {
            averages[m.name] += m.value;
            counts[m.name]++;
        }
        
        for (auto& [name, total] : averages) {
            std::cout << name << ": " << (total / counts[name]) << std::endl;
        }
    }
};

### 1. High-Level Summary
The code implements an asynchronous, multi-threaded **Event Bus** architecture used for decoupled communication between components (`DataProcessor` and `MetricsCollector`). 

- **`EventBus`**: Acts as a central dispatcher. It uses a pool of worker threads to pick up events from a `pendingEvents` queue and dispatch them to registered subscribers based on event types and priority.
- **`DataProcessor`**: A subscriber that consumes "raw" data via the `data.input` event, processes it (multiplies by 2), and then emits new events ("data.processed") back to the bus.
- **`MetricsCollector`**: A subscriber that records statistical data ("latency") received from the bus to provide summaries.
- **System Flow**: The `main` function simulates a data stream by emitting input events, which the worker threads asynchronously handle, triggering a chain reaction of processing and further event emissions.

---

### 2. Potential Bugs and Issues

1.  **Race Condition / Data Race on `subscribers` map**:
    In `EventBus::processEvents`, the worker threads access `subscribers` (via `it = subscribers.find(eventType)`) **without holding the mutex `mtx`**. While `subscribe` and `unsubscribe` correctly lock the mutex, the `processEvents` loop reads from the map while other threads might be modifying it (adding/removing handlers). This is a classic data race that can lead to undefined behavior, segmentation faults, or iterator invalidation.

2.  **Deadlock Risk (Recursive/Nested Locking)**:
    The `DataProcessor` callback for `data.input` calls `bus.emit(...)`. The `emit` function attempts to acquire `mtx`. If `processEvents` held the mutex while calling the callback (which it doesn't currently, but if a developer "fixed" the race condition by locking the mutex during the callback), the system would deadlock. Furthermore, the `DataProcessor` uses `dataMutex` and the `EventBus` uses `mtx`. If a callback in the `EventBus` tries to lock `dataMutex` while another thread holding `dataMutex` tries to call an `EventBus` method that requires `mtx`, a circular dependency could occur.

3.  **Thread Safety of `eventCounter`**:
    In `DataProcessor`, the member `eventCounter` is an `int` and is incremented inside a `bus.subscribe` lambda. Although this increment happens inside a `std::lock_guard<std::mutex> lock(dataMutex)`, it is technically safe *only* because that specific logic is wrapped in the same mutex as the data vector. However, `getEventCount()` is a naked read of `eventCounter` with **no mutex protection** and no `std::atomic`. This leads to a data race when `main` calls `getEventCount()` while a worker thread is still incrementing it.

4.  **Unsafe Destruction (Shutdown Sequence)**:
    The `EventBus` destructor sets `running = false`, but the worker threads are likely sitting in a `std::this_thread::sleep_for(std::chrono::milliseconds(1))` loop. There is no mechanism (like a `std::condition_variable`) to wake them up immediately. This means the destructor might wait for the full sleep duration, and more importantly, if there are events still in `pendingEvents` when `running` is set to `false`, they may be silently dropped because the workers exit as soon as they loop.

### 3. Performance Bottlenecks

1.  **Inefficient Sorting in the Hot Path**:
    Inside `processEvents`, for every single event processed, the code performs `std::sort` on the `handlers` vector of the specific event type. Sorting is an $O(N \log N)$ operation performed repeatedly on the same handlers. Since subscriber lists change infrequently compared to how often events are emitted, this is a massive waste of CPU cycles.
