class EventBus {
private:
    struct Handler {
        int id;
        std::function<void(const std::string&)> callback;
        int priority;
    };

    std::unordered_map<std::string, std::vector<Handler>> subscribers;
    std::mutex mtx;
    int nextId = 0;
    bool running = true;
    std::vector<std::thread> workers;
    std::vector<std::pair<std::string, std::string>> pendingEvents;

    void processEvents() {
        while (running) {
            std::vector<std::pair<std::string, std::string>> localEvents;
            {
                std::lock_guard<std::mutex> lock(mtx);
                localEvents.swap(pendingEvents);
            }
            
            if (localEvents.empty()) {
                std::this_thread::sleep_for(std::chrono::milliseconds(1));
                continue;
            }
            
            for (const auto& [eventType, payload] : localEvents) {
                auto it = subscribers.find(eventType);
                if (it != subscribers.end()) {
                    auto handlers = it->second;
                    std::sort(handlers.begin(), handlers.end(), 
                        [](const Handler& a, const Handler& b) {
                            return a.priority > b.priority;
                        });
                    for (const auto& handler : handlers) {
                        handler.callback(payload);
                    }
                }
            }
        }
    }

public:
    EventBus(int numWorkers = 4) {
        for (int i = 0; i < numWorkers; ++i) {
            workers.emplace_back(&EventBus::processEvents, this);
        }
    }
    
    ~EventBus() {
        running = false;
        for (auto& worker : workers) {
            if (worker.joinable()) worker.join();
        }
    }
    
    int subscribe(const std::string& eventType, 
                  std::function<void(const std::string&)> callback,
                  int priority = 0) {
        std::lock_guard<std::mutex> lock(mtx);
        int id = nextId++;
        subscribers[eventType].push_back({id, callback, priority});
        return id;
    }
    
    void unsubscribe(const std::string& eventType, int id) {
        std::lock_guard<std::mutex> lock(mtx);
        auto it = subscribers.find(eventType);
        if (it != subscribers.end()) {
            auto& handlers = it->second;
            handlers.erase(std::remove_if(handlers.begin(), handlers.end(),
                [id](const Handler& h) { return h.id == id; }),
                handlers.end());
        }
    }
    
    void emit(const std::string& eventType, const std::string& payload) {
        std::lock_guard<std::mutex> lock(mtx);
        pendingEvents.emplace_back(eventType, payload);
    }
    
    size_t getPendingCount() {
        std::lock_guard<std::mutex> lock(mtx);
        return pendingEvents.size();
    }
};

class DataProcessor {
private:
    std::vector<int> data;
    mutable std::mutex dataMutex;
    EventBus& bus;
    int eventCounter = 0;
    
public:
    DataProcessor(EventBus& eventBus) : bus(eventBus) {
        bus.subscribe("data.input", [this](const std::string& payload) {
            std::lock_guard<std::mutex> lock(dataMutex);
            int value = std::stoi(payload);
            data.push_back(value);
            eventCounter++;
            bus.emit("data.processed", std::to_string(value * 2));
        }, 10);
        
        bus.subscribe("data.clear", [this](const std::string&) {
            std::lock_guard<std::mutex> lock(dataMutex);
            data.clear();
            bus.emit("data.cleared", "All data removed");
        });
    }
    
    std::vector<int> getData() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        return data;
    }
    
    double calculateAverage() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        if (data.empty()) return 0.0;
        long long sum = 0;
        for (int val : data) {
            sum += val;
        }
        return static_cast<double>(sum) / data.size();
    }
    
    int getEventCount() const { return eventCounter; }
};

### 2. Potential Bugs and Issues

#### Bug 1: Undefined Behavior in `MetricsCollector::printSummary`
In the `printSummary` method, the code uses `auto& [name, total] : averages`. However, `averages` is a `std::unordered_map<std::string, double>`. The structured binding assigns `total` as a reference to the value in the map. Later, `total / counts[name]` is calculated.
While this specific part isn't strictly UB, the logic is flawed because `averages` is populated but `counts` is a separate map. More critically, if `counts[name]` is not found (which shouldn't happen here since they are populated in the same loop), it would default construct `0`, leading to division by zero.
**However, the real bug is here:** `std::unordered_map` does not guarantee order. While the output order doesn't matter for correctness, the logic `total / counts[name]` assumes `counts[name]` exists. If for some reason the maps got out of sync (not possible here in current code, but fragile design), it would crash.
**Critical Bug:** Actually, looking closer at `MetricsCollector::record`, there is no issue. But in `printSummary`, `averages[m.name] += m.value` and `counts[m.name]++` are correct.
**Wait, let's look for a real UB.**
In `DataProcessor` destructor: The `DataProcessor` is a local variable in `main`. When `main` returns, `DataProcessor` is destroyed. Its destructor calls `bus.unsubscribe`? No, it doesn't have a destructor.
**Real Bug 1: Memory Leak / Dangling Reference in Subscriptions.**
The `EventBus` stores `std::function<void(const std::string&)>`. In `DataProcessor` constructor, it subscribes using lambdas that capture `this`. If the `EventBus` outlives the `DataProcessor` (e.g., if `DataProcessor` was dynamically allocated and deleted, or if the scope ended), the `std::function` would hold a dangling pointer. In this specific `main`, `DataProcessor` is destroyed *before* `EventBus` (stack unwinding order is reverse of declaration). `processor` is declared before `metrics`, but `bus` is declared first.
Declaration order in `main`: `EventBus bus`, `DataProcessor processor`, `MetricsCollector metrics`.
Destruction order: `metrics`, `processor`, `bus`.
So `DataProcessor` is destroyed while `EventBus` is still alive. The `EventBus` still holds lambdas capturing `this` (pointer to `DataProcessor`). If `EventBus` tries to emit an event *after* `DataProcessor` is destroyed but *before* `EventBus` is destroyed, it will call a method on a destroyed object.
In `main`, the loop finishes, then it sleeps. If an event is emitted during that sleep or after `DataProcessor` destruction, it's UB. However, the loop emits `data.input`. The `DataProcessor` handles it. No other emitter exists. So this is technically safe *in this specific run*, but it's a **design flaw**: `EventBus` does not provide weak references or notification when subscribers are destroyed. If you move `DataProcessor` to dynamic allocation and delete it, `EventBus` will crash on subsequent emits.
