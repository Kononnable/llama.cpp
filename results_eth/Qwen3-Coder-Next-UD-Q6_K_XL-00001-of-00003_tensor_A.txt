class EventBus {
private:
    struct Handler {
        int id;
        std::function<void(const std::string&)> callback;
        int priority;
    };

    std::unordered_map<std::string, std::vector<Handler>> subscribers;
    std::mutex mtx;
    int nextId = 0;
    bool running = true;
    std::vector<std::thread> workers;
    std::vector<std::pair<std::string, std::string>> pendingEvents;

    void processEvents() {
        while (running) {
            std::vector<std::pair<std::string, std::string>> localEvents;
            {
                std::lock_guard<std::mutex> lock(mtx);
                localEvents.swap(pendingEvents);
            }
            
            if (localEvents.empty()) {
                std::this_thread::sleep_for(std::chrono::milliseconds(1));
                continue;
            }
            
            for (const auto& [eventType, payload] : localEvents) {
                auto it = subscribers.find(eventType);
                if (it != subscribers.end()) {
                    auto handlers = it->second;
                    std::sort(handlers.begin(), handlers.end(), 
                        [](const Handler& a, const Handler& b) {
                            return a.priority > b.priority;
                        });
                    for (const auto& handler : handlers) {
                        handler.callback(payload);
                    }
                }
            }
        }
    }

public:
    EventBus(int numWorkers = 4) {
        for (int i = 0; i < numWorkers; ++i) {
            workers.emplace_back(&EventBus::processEvents, this);
        }
    }
    
    ~EventBus() {
        running = false;
        for (auto& worker : workers) {
            if (worker.joinable()) worker.join();
        }
    }
    
    int subscribe(const std::string& eventType, 
                  std::function<void(const std::string&)> callback,
                  int priority = 0) {
        std::lock_guard<std::mutex> lock(mtx);
        int id = nextId++;
        subscribers[eventType].push_back({id, callback, priority});
        return id;
    }
    
    void unsubscribe(const std::string& eventType, int id) {
        std::lock_guard<std::mutex> lock(mtx);
        auto it = subscribers.find(eventType);
        if (it != subscribers.end()) {
            auto& handlers = it->second;
            handlers.erase(std::remove_if(handlers.begin(), handlers.end(),
                [id](const Handler& h) { return h.id == id; }),
                handlers.end());
        }
    }
    
    void emit(const std::string& eventType, const std::string& payload) {
        std::lock_guard<std::mutex> lock(mtx);
        pendingEvents.emplace_back(eventType, payload);
    }
    
    size_t getPendingCount() {
        std::lock_guard<std::mutex> lock(mtx);
        return pendingEvents.size();
    }
};

class DataProcessor {
private:
    std::vector<int> data;
    mutable std::mutex dataMutex;
    EventBus& bus;
    int eventCounter = 0;
    
public:
    DataProcessor(EventBus& eventBus) : bus(eventBus) {
        bus.subscribe("data.input", [this](const std::string& payload) {
            std::lock_guard<std::mutex> lock(dataMutex);
            int value = std::stoi(payload);
            data.push_back(value);
            eventCounter++;
            bus.emit("data.processed", std::to_string(value * 2));
        }, 10);
        
        bus.subscribe("data.clear", [this](const std::string&) {
            std::lock_guard<std::mutex> lock(dataMutex);
            data.clear();
            bus.emit("data.cleared", "All data removed");
        });
    }
    
    std::vector<int> getData() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        return data;
    }
    
    double calculateAverage() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        if (data.empty()) return 0.0;
        long long sum = 0;
        for (int val : data) {
            sum += val;
        }
        return static_cast<double>(sum) / data.size();
    }
    
    int getEventCount() const { return eventCounter; }
};

class MetricsCollector {
private:
    struct Metric {
        std::string name;
        double value;
        std::chrono::system_clock::time_point timestamp;
    };
    std::vector<Metric> metrics;
    mutable std::mutex mtx;
    
public:
    void record(const std::string& name, double value) {
        std::lock_guard<std::mutex> lock(mtx);
        metrics.push_back({name, value, std::chrono::system_clock::now()});
    }
    
    void printSummary() const {
        std::lock_guard<std::mutex> lock(mtx);
        std::unordered_map<std::string, double> averages;
        std::unordered_map<std::string, int> counts;
        
        for (const auto& m : metrics) {
            averages[m.name] += m.value;
            counts[m.name]++;
        }
        
        for (auto& [name, total] : averages) {
            std::cout << name << ": " << (total / counts[name]) << std::endl;
        }
    }
};

# Event-Driven Data Processing System Analysis

## 1. High-Level Summary

This code implements an event-driven data processing system with the following components:

- **EventBus**: A central message broker that handles event publishing and subscription with priority-based handling
- **DataProcessor**: A component that consumes "data.input" events, processes the numeric values (doubling them), and emits "data.processed" events
- **MetricsCollector**: Tracks and summarizes metrics like latency

The system demonstrates a producer-consumer pattern where events flow through the system, with the DataProcessor transforming input data and the MetricsCollector tracking performance metrics.

## 2. Potential Bugs and Issues

1. **Race condition in MetricsCollector::record()**: The timestamp is captured before the lock is acquired, which could lead to inconsistent ordering if multiple threads record metrics simultaneously.

2. **Handler copying inefficiency**: In `EventBus::processEvents()`, handlers are copied (`auto handlers = it->second;`) before sorting, which is inefficient for large handler lists.

3. **Event ordering not preserved**: The `pendingEvents` vector is swapped with an empty one in each processing cycle, but events processed in later cycles may handle data that depends on earlier events being processed first.

4. **Missing error handling**: No validation of payload data in DataProcessor (e.g., `std::stoi` could throw exceptions if payload isn't a valid integer).

5. **Dangling reference in MetricsCollector**: The lambda captures `&metrics` by reference, but if the MetricsCollector is destroyed before the EventBus, callbacks will reference invalid memory.

## 3. Performance Bottlenecks

1. **Single-threaded event processing**: All events are processed by worker threads that sequentially handle each event type, creating a bottleneck when many events of the same type accumulate.

2. **Excessive mutex contention**: The EventBus uses a single mutex for all operations, which creates contention when multiple threads emit events while workers are processing.

3. **Copy-on-access pattern**: Multiple components (`DataProcessor::getData()`, `MetricsCollector::printSummary()`) copy entire data structures while holding locks, which scales poorly with large datasets.

## 4. Suggestions for Improvement

1. **Fix race conditions**:
   - Move timestamp capture inside the lock in `MetricsCollector::record()`
   - Consider using atomic operations for the `running` flag in EventBus
   - Add thread-safe event IDs and consider event sequencing

2. **Improve performance**:
   - Use move semantics instead of copying when possible (e.g., `localEvents = std::move(pendingEvents)`)
   - Implement a lock-free queue for event publishing
   - Consider sharding the EventBus with multiple mutexes for different event types

3. **Enhance robustness**:
   - Add error handling for `std::stoi` with try/catch blocks
   - Add lifetime management for subscribers using weak pointers
   - Implement timeout mechanisms for event processing

4. **Additional improvements**:
   - Consider batch processing for better throughput
   - Add metrics for processing latency and queue depth
   - Implement graceful shutdown with acknowledgment of pending events
   - Use RAII patterns consistently for resource management
