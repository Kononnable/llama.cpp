class EventBus {
private:
    struct Handler {
        int id;
        std::function<void(const std::string&)> callback;
        int priority;
    };

    std::unordered_map<std::string, std::vector<Handler>> subscribers;
    std::mutex mtx;
    int nextId = 0;
    bool running = true;
    std::vector<std::thread> workers;
    std::vector<std::pair<std::string, std::string>> pendingEvents;

    void processEvents() {
        while (running) {
            std::vector<std::pair<std::string, std::string>> localEvents;
            {
                std::lock_guard<std::mutex> lock(mtx);
                localEvents.swap(pendingEvents);
            }
            
            if (localEvents.empty()) {
                std::this_thread::sleep_for(std::chrono::milliseconds(1));
                continue;
            }
            
            for (const auto& [eventType, payload] : localEvents) {
                auto it = subscribers.find(eventType);
                if (it != subscribers.end()) {
                    auto handlers = it->second;
                    std::sort(handlers.begin(), handlers.end(), 
                        [](const Handler& a, const Handler& b) {
                            return a.priority > b.priority;
                        });
                    for (const auto& handler : handlers) {
                        handler.callback(payload);
                    }
                }
            }
        }
    }

public:
    EventBus(int numWorkers = 4) {
        for (int i = 0; i < numWorkers; ++i) {
            workers.emplace_back(&EventBus::processEvents, this);
        }
    }
    
    ~EventBus() {
        running = false;
        for (auto& worker : workers) {
            if (worker.joinable()) worker.join();
        }
    }
    
    int subscribe(const std::string& eventType, 
                  std::function<void(const std::string&)> callback,
                  int priority = 0) {
        std::lock_guard<std::mutex> lock(mtx);
        int id = nextId++;
        subscribers[eventType].push_back({id, callback, priority});
        return id;
    }
    
    void unsubscribe(const std::string& eventType, int id) {
        std::lock_guard<std::mutex> lock(mtx);
        auto it = subscribers.find(eventType);
        if (it != subscribers.end()) {
            auto& handlers = it->second;
            handlers.erase(std::remove_if(handlers.begin(), handlers.end(),
                [id](const Handler& h) { return h.id == id; }),
                handlers.end());
        }
    }
    
    void emit(const std::string& eventType, const std::string& payload) {
        std::lock_guard<std::mutex> lock(mtx);
        pendingEvents.emplace_back(eventType, payload);
    }
    
    size_t getPendingCount() {
        std::lock_guard<std::mutex> lock(mtx);
        return pendingEvents.size();
    }
};

class DataProcessor {
private:
    std::vector<int> data;
    mutable std::mutex dataMutex;
    EventBus& bus;
    int eventCounter = 0;
    
public:
    DataProcessor(EventBus& eventBus) : bus(eventBus) {
        bus.subscribe("data.input", [this](const std::string& payload) {
            std::lock_guard<std::mutex> lock(dataMutex);
            int value = std::stoi(payload);
            data.push_back(value);
            eventCounter++;
            bus.emit("data.processed", std::to_string(value * 2));
        }, 10);
        
        bus.subscribe("data.clear", [this](const std::string&) {
            std::lock_guard<std::mutex> lock(dataMutex);
            data.clear();
            bus.emit("data.cleared", "All data removed");
        });
    }
    
    std::vector<int> getData() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        return data;
    }
    
    double calculateAverage() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        if (data.empty()) return 0.0;
        long long sum = 0;
        for (int val : data) {
            sum += val;
        }
        return static_cast<double>(sum) / data.size();
    }
    
    int getEventCount() const { return eventCounter; }
};

class MetricsCollector {

        std::string name;
        double value;
        std::chrono::system_clock::time_point timestamp;
    };
    std::vector<Metric> metrics;
    mutable std::mutex mtx;
    
public:
    void record(const std::string& name, double value) {
        std::lock_guard<std::mutex> lock(mtx);
        metrics.push_back({name, value, std::chrono::system_clock::now()});
    }
    
    void printSummary() const {
        std::lock_guard<std::mutex> lock(mtx);
        std::unordered_map<std::string, double> averages;
        std::unordered_map<std::string, int> counts;
        
        for (const auto& m : metrics) {
            averages[m.name] += m.value;
            counts[m.name]++;
        }
        
        for (auto& [name, total] : averages) {
            std::cout << name << ": " << (total / counts[name]) << std::endl;
        }
    }
};

### High-Level Summary

This system implements an event-driven architecture using an EventBus pattern where components communicate through asynchronous events. The core functionality includes:

- **EventBus**: A central message broker that allows components to subscribe to event types, emit events, and process events asynchronously using multiple worker threads
- **DataProcessor**: Processes "data.input" events by parsing the payload as an integer, storing it, and emitting "data.processed" events with the doubled value
- **MetricsCollector**: Collects timing and performance metrics from events and provides summary reporting capabilities
- **Main Application**: Demonstrates the system by emitting 100 data.input events with values 0, 10, 20, ..., 990 and then reports statistics

The system aims to provide loose coupling between components while processing events concurrently using multiple worker threads.

---

### Potential Bugs and Issues

**Issue 1: Race Condition in eventCounter**
In the `DataProcessor` class, the `eventCounter++` operation is not atomic and could lead to race conditions when multiple worker threads process events concurrently. However, since this increment is protected by `dataMutex`, this is actually not a problem.

*CORRECTION*: Actually, the increment is protected by the same mutex that protects data access, so this is not a bug.

**Issue 2: Deadlock Risk with Nested Event Emission**
When processing events, handlers might emit new events (e.g., `data.input` handler emits `data.processed`). Since the worker thread holds no lock when calling `emit`, this creates a potential deadlock scenario if a handler emits an event that needs to be processed immediately in the same thread context.

**Issue 3: No Protection Against Self-Emitted Events**
The event processing loop doesn't prevent infinite loops when an event handler emits events of the same type, potentially leading to uncontrolled recursion or infinite processing loops.

---

### Performance Bottlenecks

**Bottleneck 1: Inefficient Event Processing with Spin-Sleep**
The worker threads use a busy-waiting approach with `sleep_for(1ms)` when there are no events. This introduces latency and wastes CPU cycles. A condition variable would be much more efficient for signaling new events.

**Bottleneck 2: Sorting Handlers on Every Event Processing**
The system sorts all handlers for each event type every time an event is processed. This is inefficient, especially when many events of the same type are processed. The sorting should be done only when handlers are added or removed, not on every event dispatch.

---

### Suggestions for Improvement

1. **Replace spin-sleep with condition variables**: Use a condition variable to notify worker threads when new events are available, eliminating the 1ms polling delay and reducing CPU usage.

2. **Cache sorted handler lists**: Sort handlers only when they're added or removed, not on every event dispatch.

3. **Add queue size limits and backpressure mechanisms**: Implement bounded queues to prevent memory exhaustion under high event loads.

4. **Make nextId atomic**: While currently protected by mutex, making it `std::atomic<int>` would be cleaner.

5. **Add thread-safe event counting**: Provide thread-safe methods for getting event counts without requiring external locking.
