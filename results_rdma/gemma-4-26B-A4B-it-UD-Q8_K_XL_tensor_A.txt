class EventBus {
private:
    struct Handler {
        int id;
        std::function<void(const std::string&)> callback;
        int priority;
    };

    std::unordered_map<std::string, std::vector<Handler>> subscribers;
    std::mutex mtx;
    int nextId = 0;
    bool running = true;
    std::vector<std::thread> workers;
    std::vector<std::pair<std::string, std::string>> pendingEvents;

    void processEvents() {
        while (running) {
            std::vector<std::pair<std::string, std::string>> localEvents;
            {
                std::lock_guard<std::mutex> lock(mtx);
                localEvents.swap(pendingEvents);
            }
            
            if (localEvents.empty()) {
                std::this_thread::sleep_for(std::chrono::milliseconds(1));
                continue;
            }
            
            for (const auto& [eventType, payload] : localEvents) {
                auto it = subscribers.find(eventType);
                if (it != subscribers.end()) {
                    auto handlers = it->second;
                    std::sort(handlers.begin(), handlers.end(), 
                        [](const Handler& a, const Handler& b) {
                            return a.priority > b.priority;
                        });
                    for (const auto& handler : handlers) {
                        handler.callback(payload);
                    }
                }
            }
        }
    }

public:
    EventBus(int numWorkers = 4) {
        for (int i = 0; i < numWorkers; ++i) {
            workers.emplace_back(&EventBus::processEvents, this);
        }
    }
    
    ~EventBus() {
        running = false;
        for (auto& worker : workers) {
            if (worker.joinable()) worker.join();
        }
    }
    
    int subscribe(const std::string& eventType, 
                  std::function<void(const std::string&)> callback,
                  int priority = 0) {
        std::lock_guard<std::mutex> lock(mtx);
        int id = nextId++;
        subscribers[eventType].push_back({id, callback, priority});
        return id;
    }
    
    void unsubscribe(const std::string& eventType, int id) {
        std::lock_guard<std::mutex> lock(mtx);
        auto it = subscribers.find(eventType);
        if (it != subscribers.end()) {
            auto& handlers = it->second;
            handlers.erase(std::remove_if(handlers.begin(), handlers.end(),
                [id](const Handler& h) { return h.id == id; }),
                handlers.end());
        }
    }
    
    void emit(const std::string& eventType, const std::string& payload) {
        std::lock_guard<std::mutex> lock(mtx);
        pendingEvents.emplace_back(eventType, payload);
    }
    
    size_t getPendingCount() {
        std::lock_guard<std::mutex> lock(mtx);
        return pendingEvents.size();
    }
};

class DataProcessor {
private:
    std::vector<int> data;
    mutable std::mutex dataMutex;
    EventBus& bus;
    int eventCounter = 0;
    
public:
    DataProcessor(EventBus& eventBus) : bus(eventBus) {
        bus.subscribe("data.input", [this](const std::string& payload) {
            std::lock_guard<std::mutex> lock(dataMutex);
            int value = std::stoi(payload);
            data.push_back(value);
            eventCounter++;
            bus.emit("data.processed", std::to_string(value * 2));
        }, 10);
        
        bus.subscribe("data.clear", [this](const std::string&) {
            std::lock_guard<std::mutex> lock(dataMutex);
            data.clear();
            bus.emit("data.cleared", "All data removed");
        });
    }
    
    std::vector<int> getData() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        return data;
    }
    
    double calculateAverage() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        if (data.empty()) return 0.0;
        long long sum = 0;
        for (int val : data) {
            sum += val;
        }
        return static_cast<double>(sum) / data.size();
    }
    
    int getEventCount() const { return eventCounter; }
};

class MetricsCollector {
private:
    struct Metric {
        std::string name;
        double value;
        std::chrono::system_clock::time_point timestamp;
    };
    std::vector<Metric> metrics;
    mutable std::mutex mtx;
    
public:
    void record(const std::string& name, double value) {
        std::lock_guard<std::mutex> lock(mtx);
        metrics.push_back({name, value, std::chrono::system_clock::now()});
    }
    
    void printSummary() const {
        std::lock_guard<std::mutex> lock(mtx);
        std::unordered_map<std::string, double> averages;
        std::unordered_map<std::string, int> counts;
        
        for (const auto& m : metrics) {
            averages[m.name] += m.value;
            counts[m.name]++;
        }
        
        for (auto& [name, total] : averages) {
            std::cout << name << ": " << (total / counts[name]) << std::endl;
        }
    }
};
