class EventBus {
private:
    struct Handler {
        int id;
        std::function<void(const std::string&)> callback;
        int priority;
    };

    std::unordered_map<std::string, std::vector<Handler>> subscribers;
    std::mutex mtx;
    int nextId = 0;
    bool running = true;
    std::vector<std::thread> workers;
    std::vector<std::pair<std::string, std::string>> pendingEvents;

    void processEvents() {
        while (running) {
            std::vector<std::pair<std::string, std::string>> localEvents;
            {
                std::lock_guard<std::mutex> lock(mtx);
                localEvents.swap(pendingEvents);
            }
            
            if (localEvents.empty()) {
                std::this_thread::sleep_for(std::chrono::milliseconds(1));
                continue;
            }
            
            for (const auto& [eventType, payload] : localEvents) {
                auto it = subscribers.find(eventType);
                if (it != subscribers.end()) {
                    auto handlers = it->second;
                    std::sort(handlers.begin(), handlers.end(), 
                        [](const Handler& a, const Handler& b) {
                            return a.priority > b.priority;
                        });
                    for (const auto& handler : handlers) {
                        handler.callback(payload);
                    }
                }
            }
        }
    }

public:
    EventBus(int numWorkers = 4) {
        for (int i = 0; i < numWorkers; ++i) {
            workers.emplace_back(&EventBus::processEvents, this);
        }
    }
    
    ~EventBus() {
        running = false;
        for (auto& worker : workers) {
            if (worker.joinable()) worker.join();
        }
    }
    
    int subscribe(const std::string& eventType, 
                  std::function<void(const std::string&)> callback,
                  int priority = 0) {
        std::lock_guard<std::mutex> lock(mtx);
        int id = nextId++;
        subscribers[eventType].push_back({id, callback, priority});
        return id;
    }
    
    void unsubscribe(const std::string& eventType, int id) {
        std::lock_guard<std::mutex> lock(mtx);
        auto it = subscribers.find(eventType);
        if (it != subscribers.end()) {
            auto& handlers = it->second;
            handlers.erase(std::remove_if(handlers.begin(), handlers.end(),
                [id](const Handler& h) { return h.id == id; }),
                handlers.end());
        }
    }
    
    void emit(const std::string& eventType, const std::string& payload) {
        std::lock_guard<std::mutex> lock(mtx);
        pendingEvents.emplace_back(eventType, payload);
    }
    
    size_t getPendingCount() {
        std::lock_guard<std::mutex> lock(mtx);
        return pendingEvents.size();
    }
};

class DataProcessor {
private:
    std::vector<int> data;
    mutable std::mutex dataMutex;
    EventBus& bus;
    int eventCounter = 0;
    
public:
    DataProcessor(EventBus& eventBus) : bus(eventBus) {
        bus.subscribe("data.input", [this](const std::string& payload) {
            std::lock_guard<std::mutex> lock(dataMutex);
            int value = std::stoi(payload);
            data.push_back(value);
            eventCounter++;
            bus.emit("data.processed", std::to_string(value * 2));
        }, 10);
        
        bus.subscribe("data.clear", [this](const std::string&) {
            std::lock_guard<std::mutex> lock(dataMutex);
            data.clear();
            bus.emit("data.cleared", "All data removed");

        }

### 1. High-Level Summary

The code implements a concurrent, event-driven architecture using a custom `EventBus`.

### 2. Potential Bugs and Issues

#### Bug 1: Data Race on `metrics` vector (Critical)
The `MetricsCollector` records metrics via `bus.subscribe("data.processed", ...)`. The `data.processed` event is emitted from the `DataProcessor`'s callback, which runs inside the `EventBus` worker threads. Simultaneously, `main()` calls `metrics.printSummary()`.
*   **The Issue**: While `MetricsCollector::record` and `printSummary` both lock `mtx`, the **order of execution is not guaranteed**. However, the real bug is more subtle: `main()` calls `printSummary()` *after* a fixed sleep. If the `EventBus` workers are slow or if there's a deadlock/starvation scenario, `printSummary` might run while workers are still active.
*   **Wait, there's a bigger bug**: Look at `MetricsCollector::printSummary`. It calculates averages. But `metrics` is a `std::vector<Metric>`. The `printSummary` method iterates over `metrics`. However, `main()` does **not** wait for all pending events in the `EventBus` to be processed before calling `printSummary` or accessing `processor`.
    *   `main()` sleeps 100ms. With 100 events emitted at 1ms intervals, plus processing overhead, 100ms might be insufficient. But the code *assumes* they are done. This is a race condition: `processor.calculateAverage()` and `metrics.printSummary()` may execute before all 100 events have been processed by the `EventBus` workers. This leads to **incomplete/incorrect results**.

1.  It locks `dataMutex`.
2.  It parses `payload` and pushes to `data`.
3.  It calls `bus.emit("data.processed", ...)` **while still holding `dataMutex`**.
4.  Inside `bus.emit`, it locks the `EventBus`'s `mtx`.
5.  The `EventBus` worker threads (which might be the same thread if `numWorkers=1`, though here it's 2) will eventually run `handler.callback`.
