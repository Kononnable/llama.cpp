class EventBus {
private:
    struct Handler {
        int id;
        std::function<void(const std::string&)> callback;
        int priority;
    };

    std::unordered_map<std::string, std::vector<Handler>> subscribers;
    std::mutex mtx;
    int nextId = 0;
    bool running = true;
    std::vector<std::thread> workers;
    std::vector<std::pair<std::string, std::string>> pendingEvents;

    void processEvents() {
        while (running) {
            std::vector<std::pair<std::string, std::string>> localEvents;
            {
                std::lock_guard<std::mutex> lock(mtx);
                localEvents.swap(pendingEvents);
            }
            
            if (localEvents.empty()) {
                std::this_thread::sleep_for(std::chrono::milliseconds(1));
                continue;
            }
            
            for (const auto& [eventType, payload] : localEvents) {
                auto it = subscribers.find(eventType);
                if (it != subscribers.end()) {
                    auto handlers = it->second;
                    std::sort(handlers.begin(), handlers.end(), 
                        [](const Handler& a, const Handler& b) {
                            return a.priority > b.priority;
                        });
                    for (const auto& handler : handlers) {
                        handler.callback(payload);
                    }
                }
            }
        }
    }

public:
    EventBus(int numWorkers = 4) {
        for (int i = 0; i < numWorkers; ++i) {
            workers.emplace_back(&EventBus::processEvents, this);
        }
    }
    
    ~EventBus() {
        running = false;
        for (auto& worker : workers) {
            if (worker.joinable()) worker.join();
        }
    }
    
    int subscribe(const std::string& eventType, 
                  std::function<void(const std::string&)> callback,
                  int priority = 0) {
        std::lock_guard<std::mutex> lock(mtx);
        int id = nextId++;
        subscribers[eventType].push_back({id, callback, priority});
        return id;
    }
    
    void unsubscribe(const std::string& eventType, int id) {
        std::lock_guard<std::mutex> lock(mtx);
        auto it = subscribers.find(eventType);
        if (it != subscribers.end()) {
            auto& handlers = it->second;
            handlers.erase(std::remove_if(handlers.begin(), handlers.end(),
                [id](const Handler& h) { return h.id == id; }),
                handlers.end());
        }
    }
    
    void emit(const std::string& eventType, const std::string& payload) {
        std::lock_guard<std::mutex> lock(mtx);
        pendingEvents.emplace_back(eventType, payload);
    }
    
    size_t getPendingCount() {
        std::lock_guard<std::mutex> lock(mtx);
        return pendingEvents.size();
    }
};

class DataProcessor {
private:
    std::vector<int> data;
    mutable std::mutex dataMutex;
    EventBus& bus;
    int eventCounter = 0;
    
public:
    DataProcessor(EventBus& eventBus) : bus(eventBus) {
        bus.subscribe("data.input", [this](const std::string& payload) {
            std::lock_guard<std::mutex> lock(dataMutex);
            int value = std::stoi(payload);
            data.push_back(value);
            eventCounter++;
            bus.emit("data.processed", std::to_string(value * 2));
        }, 10);
        
        bus.subscribe("data.clear", [this](const std::string&) {
            std::lock_guard<std::mutex> lock(dataMutex);
            data.clear();
            bus.emit("data.cleared", "All data removed");
        });
    }
    
    std::vector<int> getData() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        return data;
    }
    
    double calculateAverage() const {
        std::lock_guard<std::mutex> lock(dataMutex);
        if (data.empty()) return 0.0;
        long long sum = 0;
        for (int val : data) {
            sum += val;
        }
        return static_cast<double>(sum) / data.size();
    }
    
    int getEventCount() const { return eventCounter; }
};

class MetricsCollector {
private:
    struct Metric {
        std::string name;
        double value;
        std::chrono::system_clock::time_point timestamp;
    };
    std::vector<Metric> metrics;
    mutable std::mutex mtx;
    
public:
    void record(const std::string& name, double value) {
        std::lock_guard<std::mutex> lock(mtx);
        metrics.push_back({name, value, std::chrono::system_clock::now()});
    }
    
    void printSummary() const {
        std::lock_guard<std::mutex> lock(mtx);
        std::unordered_map<std::string, double> averages;
        std::unordered_map<std::string, int> counts;
        
        for (const auto& m : metrics) {
            averages[m.name] += m.value;
            counts[m.name]++;
        }
        
        for (auto& [name, total] : averages) {
            std::cout << name << ": " << (total / counts[name]) << std::endl;
        }
    }
};

### 1. **High-Level Summary**  
This system implements an **event-driven architecture** with a custom `EventBus` for asynchronous message passing, a `DataProcessor` that reacts to events by transforming and storing data, and a `MetricsCollector` for recording and summarizing metrics. Key behaviors include:  
- Multi-threaded event processing (via `EventBus` workers).  
- Event subscriptions (e.g., `data.input` triggers value doubling, `data.clear` resets state).  
- Metrics collection (e.g., latency from processed data).  
- Simulated workload: emitting 100 `data.input` events with small delays.

### 2. **Potential Bugs/Issues**  
**a. Race Condition in `EventBus::processEvents()`**  
- **Issue**: `localEvents` is swapped into `pendingEvents` under lock, but handlers are copied *outside* the lock (`auto handlers = it->second`). If `unsubscribe()` modifies the original `subscribers[eventType]` during processing, the copy may become invalid or inconsistent.  
- **Fix**: Clone handlers *while holding the lock*, or avoid copying by iterating over a reference (but require thread-safety guarantees).  

**b. Undefined Behavior in `std::stoi`/`std::stod` Without Validation**  
- **Issue**: `std::stoi(payload)` in `DataProcessor` and `MetricsCollector` crashes if `payload` is non-numeric (e.g., `bus.emit("data.clear", "All data removed")` → `stoi("All data removed")`).  
- **Fix**: Validate input format or use `std::optional`/error handling.  

**c. `running = false` Not Thread-Safe in Destructor**  
- **Issue**: `running` is read in `processEvents()` without synchronization (only written in the destructor). The compiler may optimize away re-reading due to missing `std::atomic`.  
- **Fix**: Declare `running` as `std::atomic<bool>`.  

---

### 3. **Performance Bottlenecks**  
**a. Excessive Lock Contention on `EventBus::mtx`**  
- **Issue**: All operations (`subscribe`, `emit`, `processEvents`) contend for `EventBus::mtx`, serializing event dispatch and subscriber management. The `processEvents()` loop acquires the lock *per batch* but still blocks producers.  
- **Mitigation**: Use lock-free queues (e.g., `std::atomic` queues) or partition the `EventBus` by topic.  

**b. Unnecessary Copying in `processEvents()`**  
- **Issue**: `localEvents.swap(pendingEvents)` copies the entire `pendingEvents` vector. With high throughput, this becomes costly.  
- **Mitigation**: Use move semantics or a shared queue (e.g., `concurrent_queue`).  

---

### 4. **Suggestions for Improvement**  
- **Thread-Safety**: Replace `std::mutex` with `std::atomic` for `running`, and use `std::shared_mutex` for `subscribers` to allow concurrent reads.  
- **Resource Cleanup**: Add `unsubscribe` on destruction (e.g., store handler IDs in `DataProcessor` and clean up in its destructor).  
- **Scalability**: Replace `std::vector` in `subscribers` with a sorted structure (e.g., `std::priority_queue`) to avoid per-event sorting.  
- **Error Resilience**: Wrap event callbacks in `try/catch` to prevent exceptions from crashing worker threads.  
- **Metrics**: Add timestamped snapshots to `MetricsCollector::printSummary()` for trend analysis.  

---
