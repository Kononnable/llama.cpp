class EventBus {
private:
    struct Handler {
        int id;
        std::function<void(const std::string&)> callback;
        int priority;
    };

    std::unordered_map<std::string, std::vector<Handler>> subscribers;
    std::mutex mtx;
    int nextId = 0;
    bool running = true;
    std::vector<std::thread> workers;
    std::vector<std::pair<std::string, std::string>> pendingEvents;

    void processEvents() {
        while (running) {
            std::vector<std::pair<std::string, std::string>> localEvents;
            {
                std::lock_guard<std::mutex> lock(mtx);
                localEvents.swap(pendingEvents);
            }
            
            if (localEvents.empty()) {
                std::this_thread::sleep_for(std::chrono::milliseconds(1));
                continue;
            }
            
            for (const auto& [eventType, payload] : localEvents) {
                auto it = subscribers.find(eventType);
                if (it != subscribers.end()) {
                    auto handlers = it->second;
                    std::sort(handlers.begin(), handlers.end(), 
                        [](const Handler& a, const Handler& b) {
                            return a.priority > b.priority;
                        });
                    for (const auto& handler : handlers) {
                        handler.callback(payload);
                    }
                }
            }
        }
    }

public:
    EventBus(int numWorkers = 4) {
        for (int i = 0; i < numWorkers; ++i) {
            workers.emplace_back(&EventBus::processEvents, this);
        }
    }
    
    ~EventBus() {
        running = false;
        for (auto& worker : workers) {
            if (worker.joinable()) worker.join();
        }
    }
    
    int subscribe(const std::string& eventType, 
                  std::function<void(const std::string&)> callback,
                  int priority = 0) {
        std::lock_guard<std::mutex> lock(mtx);
        int id = nextId++;
        subscribers[eventType].push_back({id, callback, priority});
        return id;
    }
    
    void unsubscribe(const std::string& eventType, int id) {
        std::lock_guard<std::mutex> lock(mtx);
        auto it = subscribers.find(eventType);
        if (it != subscribers.end()) {
            auto& handlers = it->second;
            handlers.erase(std::remove_if(handlers.begin(), handlers.end(),
                [id](const Handler& h) { return h.id == id; }),
                handlers.end());
        }
    }
    
    void emit(const std::string& eventType, const std::string& payload) {
        std::lock_guard<std::mutex> lock(mtx);
        pendingEvents.emplace_back(eventType, payload);
    }
    
    size_t getPendingCount() {
        std::lock_guard<std::mutex> lock(mtx);
        return pendingEvents.size();
    }
};

class DataProcessor {

            bus.emit("data.processed", std::to_string(value * 2));

### 2. Potential Bugs and Issues

1.  **Race Condition in `MetricsCollector::printSummary` (Critical Bug)**
    *   **Issue**: In `MetricsCollector::printSummary`, the method locks the mutex, iterates over `metrics`, and computes averages into temporary unordered maps (`averages`, `counts`). However, `metrics` is a `std::vector` that can be modified concurrently by `record()`. While the lock protects the push_back, the iteration logic assumes `metrics` is static during the calculation. More importantly, if `printSummary` is called while another thread is calling `record`, the lock ensures safety for `record`, but the iteration inside `printSummary` holds the lock. This is actually *safe* from a data corruption standpoint, **BUT**:
    *   **Subtle Bug**: If `record` is called concurrently, it locks the mutex. If `printSummary` is called concurrently, it also locks the mutex. This is fine. However, look at the `averages` and `counts` maps. They are local variables. The code does `averages[m.name] += m.value`. This is safe.

If a worker thread throws an exception (e.g., `std::stoi` fails because payload is not an integer), the worker thread will terminate abruptly without joining, causing `main` to hang or crash later when joining. The code assumes all payloads are valid integers.
